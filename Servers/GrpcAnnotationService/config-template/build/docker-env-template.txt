# --- Identity Server (token issuer/introspection) ---
# Option A (default): throwaway test Identity Server, see
# Servers/IdentityServer/DevTest/Config.cs. "host.docker.internal" is used
# instead of "identity-devtest" so the issuer matches whether the token was
# requested by a host-side test process (localhost:5020) or validated from
# inside this container.

IdentityServer__ClientSecret=DevTestIntrospectionSecret
