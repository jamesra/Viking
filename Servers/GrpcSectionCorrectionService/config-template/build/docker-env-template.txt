IdentityServer__ClientSecret=REPLACE_WITH_CORRECTION_SERVICE_CLIENT_SECRET
CorrectionService__Name=REPLACE_WITH_CORRECTION_SERVICE_NAME

# Fallback when IdentityServer__* is not set: read the Identity SQL directly (lists Resource rows with
# ResourceTypeId AnnotationContext). Database name must match Identity Server SQL_SERVER_IDENTITY_DB (IdentityViking).
ConnectionStrings__IdentityConnection=Server=192.168.0.110,1433;Database=IdentityViking;User ID=VikingDirect;Password=REPLACE_WITH_SQL_PASSWORD;TrustServerCertificate=True
